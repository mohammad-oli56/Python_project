import tkinter as tk
from tkinter import font

class TodoApp:
    def __init__(self, root):
        self.root = root
        self.root.title("My Tasks")
        self.root.geometry("450x700")
        self.root.configure(bg="#FAF8F5")  # Warm cream background

        # Colors (matching the React version)
        self.bg_color = "#FAF8F5"
        self.card_bg = "#FFFFFF"
        self.primary = "#E07A5F"  # Warm coral
        self.primary_hover = "#C96A52"
        self.text_dark = "#2D3142"
        self.text_muted = "#9CA3AF"
        self.success = "#81B29A"

        # Fonts
        self.title_font = font.Font(family="Georgia", size=28, weight="bold")
        self.subtitle_font = font.Font(family="Arial", size=11)
        self.task_font = font.Font(family="Arial", size=13)
        self.strike_font = font.Font(family="Arial", size=13, overstrike=True)
        self.btn_font = font.Font(family="Arial", size=13, weight="bold")

        self.tasks = []

        # --- Header ---
        header_frame = tk.Frame(root, bg=self.bg_color)
        header_frame.pack(pady=(40, 5))
        
        tk.Label(header_frame, text="My Tasks",
                 font=self.title_font,
                 bg=self.bg_color, fg=self.text_dark).pack()

        tk.Label(root, text="Stay organized, one task at a time",
                 font=self.subtitle_font,
                 bg=self.bg_color, fg=self.text_muted).pack(pady=(0, 25))

        # --- Input Container ---
        input_container = tk.Frame(root, bg=self.card_bg, bd=0,
                                   highlightthickness=2, highlightbackground="#E5E5E5")
        input_container.pack(pady=10, padx=30, fill="x")

        self.task_entry = tk.Entry(input_container, font=self.task_font,
                                   bd=0, bg=self.card_bg, fg=self.text_dark,
                                   insertbackground=self.primary)
        self.task_entry.insert(0, "What needs to be done?")
        self.task_entry.config(fg=self.text_muted)
        self.task_entry.pack(side="left", fill="x", expand=True, padx=15, pady=12)
        self.task_entry.bind("<FocusIn>", self.on_entry_focus_in)
        self.task_entry.bind("<FocusOut>", self.on_entry_focus_out)
        self.task_entry.bind("<Return>", lambda e: self.add_task())

        add_btn = tk.Button(input_container, text="+",
                            bg=self.primary, fg="white",
                            font=("Arial", 16, "bold"),
                            command=self.add_task,
                            cursor="hand2", bd=0,
                            width=3, height=1,
                            activebackground=self.primary_hover,
                            activeforeground="white")
        add_btn.pack(side="right", padx=8, pady=8)
        self.add_hover(add_btn, self.primary_hover, self.primary)

        # --- Task Container with Scrollbar ---
        self.outer_frame = tk.Frame(root, bg=self.bg_color)
        self.outer_frame.pack(pady=20, padx=30, fill="both", expand=True)

        self.canvas = tk.Canvas(self.outer_frame, bg=self.bg_color, 
                                highlightthickness=0, bd=0)
        self.task_list_frame = tk.Frame(self.canvas, bg=self.bg_color)
        self.scrollbar = tk.Scrollbar(self.outer_frame, orient="vertical", 
                                       command=self.canvas.yview)
        self.canvas.configure(yscrollcommand=self.scrollbar.set)

        self.scrollbar.pack(side="right", fill="y")
        self.canvas.pack(side="left", fill="both", expand=True)
        self.canvas_window = self.canvas.create_window((0, 0), 
                                                        window=self.task_list_frame, 
                                                        anchor="nw")

        self.task_list_frame.bind("<Configure>", self.on_frame_configure)
        self.canvas.bind("<Configure>", self.on_canvas_configure)

        # --- Empty State ---
        self.empty_frame = tk.Frame(self.task_list_frame, bg=self.bg_color)
        self.empty_label = tk.Label(self.empty_frame, text="📋",
                                    font=("Arial", 48), bg=self.bg_color)
        self.empty_label.pack(pady=(50, 10))
        tk.Label(self.empty_frame, text="No tasks yet",
                 font=("Georgia", 16, "bold"),
                 bg=self.bg_color, fg=self.text_dark).pack()
        tk.Label(self.empty_frame, text="Add your first task above to get started",
                 font=self.subtitle_font,
                 bg=self.bg_color, fg=self.text_muted).pack(pady=5)
        self.empty_frame.pack(fill="x")

        # --- Stats Footer ---
        self.stats_frame = tk.Frame(root, bg=self.bg_color)
        self.stats_frame.pack(side="bottom", fill="x", pady=20, padx=30)

        self.remaining_label = tk.Label(self.stats_frame, 
                                        text="0 tasks remaining",
                                        bg=self.bg_color, fg=self.text_muted, 
                                        font=self.subtitle_font)
        self.remaining_label.pack(side="left")

        self.clear_btn = tk.Button(self.stats_frame, text="Clear completed",
                                   bg=self.bg_color, fg=self.primary,
                                   font=self.subtitle_font,
                                   command=self.delete_completed,
                                   cursor="hand2", bd=0,
                                   activebackground=self.bg_color,
                                   activeforeground=self.primary_hover)
        self.clear_btn.pack(side="right")

    def on_frame_configure(self, event):
        self.canvas.config(scrollregion=self.canvas.bbox("all"))

    def on_canvas_configure(self, event):
        self.canvas.itemconfig(self.canvas_window, width=event.width)

    def on_entry_focus_in(self, event):
        if self.task_entry.get() == "What needs to be done?":
            self.task_entry.delete(0, tk.END)
            self.task_entry.config(fg=self.text_dark)

    def on_entry_focus_out(self, event):
        if not self.task_entry.get():
            self.task_entry.insert(0, "What needs to be done?")
            self.task_entry.config(fg=self.text_muted)

    def add_hover(self, widget, hover_color, normal_color):
        def on_enter(e): widget.config(bg=hover_color)
        def on_leave(e): widget.config(bg=normal_color)
        widget.bind("<Enter>", on_enter)
        widget.bind("<Leave>", on_leave)

    def add_task(self):
        text = self.task_entry.get().strip()
        if text and text != "What needs to be done?":
            # Hide empty state
            self.empty_frame.pack_forget()

            # Create task card
            row = tk.Frame(self.task_list_frame, bg=self.card_bg, bd=0,
                          highlightthickness=1, highlightbackground="#E5E5E5")
            row.pack(fill="x", pady=4)

            inner = tk.Frame(row, bg=self.card_bg)
            inner.pack(fill="x", padx=15, pady=12)

            var = tk.BooleanVar()

            # Custom checkbox using canvas
            check_canvas = tk.Canvas(inner, width=22, height=22, 
                                     bg=self.card_bg, highlightthickness=0,
                                     cursor="hand2")
            check_canvas.pack(side="left", padx=(0, 12))
            
            circle = check_canvas.create_oval(2, 2, 20, 20, outline="#D1D5DB", width=2)
            checkmark = check_canvas.create_text(11, 11, text="✓", 
                                                  font=("Arial", 10, "bold"),
                                                  fill="white", state="hidden")

            label = tk.Label(inner, text=text, bg=self.card_bg,
                            font=self.task_font, anchor="w", fg=self.text_dark)
            label.pack(side="left", fill="x", expand=True)

            def toggle(event=None):
                var.set(not var.get())
                self.update_checkbox(var, check_canvas, circle, checkmark, label)
                self.update_stats()

            check_canvas.bind("<Button-1>", toggle)
            label.bind("<Button-1>", toggle)

            self.tasks.append({
                "var": var, 
                "frame": row, 
                "label": label,
                "canvas": check_canvas,
                "circle": circle,
                "checkmark": checkmark
            })
            
            self.task_entry.delete(0, tk.END)
            self.update_stats()

    def update_checkbox(self, var, canvas, circle, checkmark, label):
        if var.get():
            canvas.itemconfig(circle, fill=self.success, outline=self.success)
            canvas.itemconfig(checkmark, state="normal")
            label.config(font=self.strike_font, fg=self.text_muted)
        else:
            canvas.itemconfig(circle, fill="", outline="#D1D5DB")
            canvas.itemconfig(checkmark, state="hidden")
            label.config(font=self.task_font, fg=self.text_dark)

    def delete_completed(self):
        new_tasks = []
        for task in self.tasks:
            if task["var"].get():
                task["frame"].destroy()
            else:
                new_tasks.append(task)
        self.tasks = new_tasks
        self.update_stats()
        
        if not self.tasks:
            self.empty_frame.pack(fill="x")

    def update_stats(self):
        total = len(self.tasks)
        completed = sum(1 for t in self.tasks if t["var"].get())
        remaining = total - completed
        self.remaining_label.config(text=f"{remaining} task{'s' if remaining != 1 else ''} remaining")

if __name__ == "__main__":
    root = tk.Tk()
    app = TodoApp(root)
    root.mainloop()
